3. redis

# 重启服务
先# 1. 先重启从节点（6380, 6381）
# 2. 最后重启主节点（6379）
# 这样可以最小化对服务的影响
sudo systemctl restart redis-6380 

# 等待几秒，确认启动成功
sleep 3

# 检查状态
sudo systemctl status redis-6380 --no-pager -l

# 测试连接
redis-cli -h 192.168.10.14 -p 6380 ping

# 查看是否有错误
sudo journalctl -u redis-6380 -n 20 --no-pager


redis-cli -p 6379 cluster info
redis-cli -p 6379 cluster nodes

# 集群模式连接（-c 必须加，否则无法跨分片操作）
redis-cli -c -h 192.168.10.14 -p 6379

redis-cli -p 6379 CONFIG GET maxclients
