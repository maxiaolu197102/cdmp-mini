1. 获取令牌,并创建一个用户
ADMIN_TOKEN=$(curl -s -H 'Content-Type: application/json' \
  -d '{"username":"admin","password":"Admin@2021"}' \
  http://127.0.0.1:8088/login | jq -r '.data.access_token')

  curl -s -X POST http://127.0.0.1:8088/v1/users \
  -H "Authorization: Bearer $ADMIN_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
        "metadata": {"name": "demo-user-001"},
        "nickname": "Demo User",
        "password": "Passw0rd!",
        "email": "demo-user-001@example.com",
        "phone": "+8613800138000",
        "status": 1,
        "isAdmin": 0
      }'
